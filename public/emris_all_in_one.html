Structure SQL de la table utilisateurs

Exécutez ce script une seule fois dans le SQL Editor de Supabase

CREATE TABLE IF NOT EXISTS public.utilisateurs (
  id TEXT PRIMARY KEY DEFAULT ('usr-' || floor(extract(epoch from now()) * 1000)::text),
  nom TEXT NOT NULL,
  telephone TEXT,
  email TEXT UNIQUE NOT NULL,
  mot_de_passe TEXT,
  role TEXT DEFAULT 'client',
  account_status TEXT DEFAULT 'pending',
  subscription_plan TEXT DEFAULT 'Essai Gratuit 7 Jours',
  subscription_status TEXT DEFAULT 'trial',
  payment_status TEXT DEFAULT 'none',
  total_paid_fcfa NUMERIC DEFAULT 0,
  device TEXT DEFAULT 'Formulaire Interne Emris',
  notes TEXT,
  registered_at TIMESTAMPTZ DEFAULT NOW(),
  last_login_at TIMESTAMPTZ DEFAULT NOW(),
  created_at TIMESTAMPTZ DEFAULT NOW()
);

-- Sécurité RLS et politiques
ALTER TABLE public.utilisateurs ENABLE ROW LEVEL SECURITY;

CREATE POLICY "Lecture publique utilisateurs"
  ON public.utilisateurs FOR SELECT
  USING (true);

CREATE POLICY "Insertion formulaire utilisateurs"
  ON public.utilisateurs FOR INSERT
  WITH CHECK (true);

CREATE POLICY "Modification utilisateurs"
  ON public.utilisateurs FOR UPDATE
  USING (true);

CREATE POLICY "Suppression utilisateurs"
  ON public.utilisateurs FOR DELETE
  USING (true);
